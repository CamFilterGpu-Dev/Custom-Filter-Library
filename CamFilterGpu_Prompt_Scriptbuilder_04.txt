import cv2
import numpy as np
import traceback

class CustomFilter:
    def __init__(self):
        self.name = "Vignette & Tint"
        self.enabled = True
        self.mode = "Both"        # "Vignette", "Tint", or "Both"
        self.strength = 0.5
        self.tint = "#FFB74D"

    def release(self):
        pass

    def get_ui_schema(self):
        return [
            {"type": "label", "label": "Vignette & Tint", "font_size": 13, "row": 0, "column": 0, "columnspan": 4},
            {"type": "switch", "name": "enabled", "label": "Enable", "default": True, "row": 1, "column": 0, "columnspan": 4},
            {"type": "dropdown", "name": "mode", "label": "Mode", "options": ["Vignette", "Tint", "Both"], "default": "Both", "row": 2, "column": 0, "columnspan": 4},
            {"type": "slider", "name": "strength", "label": "Strength", "min": 0.0, "max": 1.0, "default": 0.5, "row": 3, "column": 0, "columnspan": 4},
            {"type": "color", "name": "tint", "label": "Tint", "default": "#FFB74D", "row": 4, "column": 0, "columnspan": 4},
            {"type": "button", "name": "reset_defaults", "label": "Reset", "row": 5, "column": 0, "columnspan": 4},
            {"type": "gap", "row": 6, "column": 0, "columnspan": 4},
        ]

    def reset_defaults(self):
        self.enabled = True
        self.mode = "Both"
        self.strength = 0.5
        self.tint = "#FFB74D"

    def hex_to_bgr(self, hex_val):
        hex_val = str(hex_val).lstrip('#')
        if len(hex_val) != 6:
            return (255, 255, 255)
        r, g, b = tuple(int(hex_val[i:i+2], 16) for i in (0, 2, 4))
        return (b, g, r)

    def apply(self, frame, mask, hands, eyes):
        if frame is None or frame.size == 0:
            return frame

        try:
            enabled = bool(getattr(self, 'enabled', True))
            mode = str(getattr(self, 'mode', 'Both'))
            strength = float(getattr(self, 'strength', 0.5))

            if not enabled or strength <= 0.0:
                return frame

            h, w = frame.shape[:2]

            # Vignette — one vectorized radial gradient, no Python loops.
            if mode in ("Vignette", "Both"):
                Y, X = np.ogrid[:h, :w]
                cx, cy = w / 2.0, h / 2.0
                dist = np.sqrt(((X - cx) / max(w / 2.0, 1.0)) ** 2 +
                               ((Y - cy) / max(h / 2.0, 1.0)) ** 2)
                vignette = np.clip(1.0 - strength * dist, 0.0, 1.0).astype(np.float32)
                frame = (frame.astype(np.float32) * vignette[..., None]).clip(0, 255).astype(np.uint8)

            # Tint — blend toward a solid color.
            if mode in ("Tint", "Both"):
                tint_bgr = self.hex_to_bgr(getattr(self, 'tint', '#FFB74D'))
                tint_layer = np.full((h, w, 3), tint_bgr, dtype=np.uint8)
                frame = cv2.addWeighted(frame, 1.0 - strength, tint_layer, strength, 0)

            return frame.astype(np.uint8)
        except Exception as e:
            print(f"Filter Error Bypassed: {e}")
            traceback.print_exc()
            return frame
